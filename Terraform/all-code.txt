---------

Different configurations for different instances 

provider "aws" {
  region = "ap-south-1"
}

locals {
  instance_types = {
    dev = "t3.micro"
    uat = "t3.small"
  }
}

resource "aws_instance" "webserver" {

  for_each = local.instance_types

  ami           = "ami-04394a3a96907e9ec"
  instance_type = each.value
  tags = {
    Name = "WebServer-${each.key}"
  }
}

connection and remote-exec option to install commands inside ec2 instance

provider "aws" {
   region = "ap-south-1"
 }

 resource "aws_instance" "mumbaiserver" {
   ami           = "ami-088fb12f8e8397f06"
   instance_type = "t3.small"
   vpc_security_group_ids = ["sg-0520819dbbf4e48a9"]
   key_name = "awar11-na"

   connection {
     type = "ssh"
     user = "ec2-user"
     private_key = file("/Users/avizway/Desktop/keypairs/awar11-na.pem")
     host = self.public_ip
   }

   provisioner "remote-exec" {
     inline = [
       "sudo dnf install httpd -y",
       "sudo systemctl start httpd",
       "sudo systemctl enable httpd",
       "echo '<h1>Hello from terraform remoteexec SSH option</h1>' | sudo tee /var/www/html/index.html"
     ]
   }

   tags = {
     Name = "Web-server"
   }
 }

SAME AS ABOVE, but USING DYNAMIC BLOCK

provider "aws" {
  region = "ap-south-1"
}

data "aws_vpc" "Existingvpc" {
  filter {
    name   = "tag:Name"
    values = ["project-vpc"]
  }
}

data "aws_subnet" "public-1a" {
  filter {
    name   = "tag:Name"
    values = ["project-subnet-public1-ap-south-1a"]
  }
}

data "aws_subnet" "public-1b" {
  filter {
    name   = "tag:Name"
    values = ["project-subnet-public2-ap-south-1b"]
  }
}

data "aws_ami" "al2023" {
  most_recent = true
  owners      = ["amazon"]

  filter {
    name   = "name"
    values = ["al2023-ami-2023*-kernel-6.18-x86_64"]
  }
  filter {
    name   = "description"
    values = ["Amazon Linux 2023*"]
  }
}

locals {
  ingress_rules = [
    { port = 22 },
    { port = 80 },
    { port = 443 },
    { port = 8080 }
  ]
}

resource "aws_security_group" "manual_example" {
  name = "Manual-example"
  description = "Custom VPC SG for Web Server"
  vpc_id = data.aws_vpc.Existingvpc.id

  dynamic "ingress" {
    for_each = local.ingress_rules
    content {
      from_port = ingress.value.port
      to_port = ingress.value.port
      protocol = "tcp"
      cidr_blocks = [ "0.0.0.0/0" ]
    }
  }
}

resource "aws_instance" "mumbaiserver" {
  ami           = data.aws_ami.al2023.id
  instance_type = "t3.small"
  subnet_id     = data.aws_subnet.public-1a.id
  vpc_security_group_ids = [ aws_security_group.manual_example.id ]
  key_name      = "awar11-na"

  tags = {
    Name = "Web-server"
  }
}
